def native_windows():
    windows=[]
    def info(hwnd):
        name=ctypes.create_unicode_buffer(256);user.GetClassNameW(hwnd,name,256)
        rect=wintypes.RECT();user.GetWindowRect(hwnd,ctypes.byref(rect))
        point=wintypes.POINT();user.ClientToScreen(hwnd,ctypes.byref(point))
        client=wintypes.RECT();user.GetClientRect(hwnd,ctypes.byref(client))
        return {'hwnd':int(hwnd),'class':name.value,'rect':{'x':rect.left,'y':rect.top,'width':rect.right-rect.left,'height':rect.bottom-rect.top},'client_origin':{'x':point.x,'y':point.y},'client_size':{'width':client.right,'height':client.bottom},'dpi':user.GetDpiForWindow(hwnd)}
    def each(hwnd,_):
        row=info(hwnd)
        if row['class']=='Chrome_WidgetWin_1' and user.IsWindowVisible(hwnd):
            kids=[]
            def child(ch,_):
                r=info(ch)
                if r['class']=='Chrome_RenderWidgetHostHWND':kids.append(r)
                return True
            user.EnumChildWindows(hwnd,CB(child),0);row['render_children']=kids;windows.append(row)
        return True
    user.EnumWindows(CB(each),0)
    return windows
