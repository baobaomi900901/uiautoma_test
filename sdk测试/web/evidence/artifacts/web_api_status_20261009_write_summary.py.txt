def qualified(row):
    api = row['api']
    if api.startswith(('WebElement.', 'WebBrowser.')):
        return api
    if 'WebElement' in row['group']:
        return 'WebElement.'+api
    if 'WebBrowser' in row['group']:
        return 'WebBrowser.'+api
    return 'uiautoma.web.'+api
